🚀 Cloud Sync & Tools

Sync all locally registered pets, volunteers, consents, feedback, and enquiries directly to your live Supabase database with one click.

💡 Offline / Demo Resilient:
When Supabase is connected, all records are stored in your cloud database and backed up locally. If disconnected, it safely falls back to LocalStorage so no demo fails.

📄 Supabase SQL Schema

To create the required tables in your Supabase project, execute this script in your Supabase Dashboard > SQL Editor (also available in supabase_schema.sql): 

-- Run in Supabase SQL Editor:
CREATE TABLE IF NOT EXISTS public.pets (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  pet_name TEXT NOT NULL, species TEXT NOT NULL DEFAULT 'Dog', breed TEXT,
  age TEXT, gender TEXT, owner_name TEXT NOT NULL, owner_phone TEXT NOT NULL,
  owner_email TEXT, emergency_contact TEXT, diet_notes TEXT, medical_notes TEXT,
  stay_start TEXT, stay_end TEXT, status TEXT DEFAULT 'Active', room_number TEXT,
  created_at TIMESTAMPTZ DEFAULT NOW()
);
CREATE TABLE IF NOT EXISTS public.volunteers (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  full_name TEXT NOT NULL, email TEXT NOT NULL, phone TEXT NOT NULL,
  availability TEXT, activities TEXT, experience TEXT, motivation TEXT,
  status TEXT DEFAULT 'Pending', created_at TIMESTAMPTZ DEFAULT NOW()
);
CREATE TABLE IF NOT EXISTS public.consents (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  owner_name TEXT NOT NULL, pet_name TEXT NOT NULL, owner_phone TEXT NOT NULL,
  owner_email TEXT, emergency_vet_consent BOOLEAN DEFAULT TRUE,
  terms_accepted BOOLEAN DEFAULT TRUE, signature_name TEXT NOT NULL,
  signature_date TEXT, status TEXT DEFAULT 'Signed', created_at TIMESTAMPTZ DEFAULT NOW()
);
CREATE TABLE IF NOT EXISTS public.feedback (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  author_name TEXT NOT NULL, pet_name TEXT NOT NULL, overall_rating INTEGER DEFAULT 5,
  cleanliness_rating INTEGER DEFAULT 5, staff_rating INTEGER DEFAULT 5, care_rating INTEGER DEFAULT 5,
  comments TEXT, recommend TEXT DEFAULT 'Yes', created_at TIMESTAMPTZ DEFAULT NOW()
);
CREATE TABLE IF NOT EXISTS public.enquiries (
  id BIGINT GENERATED BY DEFAULT AS IDENTITY PRIMARY KEY,
  name TEXT NOT NULL, email TEXT NOT NULL, phone TEXT, pet_type TEXT,
  message TEXT NOT NULL, status TEXT DEFAULT 'New', created_at TIMESTAMPTZ DEFAULT NOW()
);
ALTER TABLE public.pets ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.volunteers ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.consents ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.feedback ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.enquiries ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Public insert pets" ON public.pets FOR INSERT WITH CHECK (true);
CREATE POLICY "Public read volunteers" ON public.volunteers FOR SELECT USING (true);
CREATE POLICY "Public insert volunteers" ON public.volunteers FOR INSERT WITH CHECK (true);
CREATE POLICY "Public read consents" ON public.consents FOR SELECT USING (true);
CREATE POLICY "Public insert consents" ON public.consents FOR INSERT WITH CHECK (true);
CREATE POLICY "Public read feedback" ON public.feedback FOR SELECT USING (true);
CREATE POLICY "Public insert feedback" ON public.feedback FOR INSERT WITH CHECK (true);
CREATE POLICY "Public read enquiries" ON public.enquiries FOR SELECT USING (true);
CREATE POLICY "Public insert enquiries" ON public.enquiries FOR INSERT WITH CHECK (true);
